cmake_minimum_required(VERSION 3.16)
project(nexus_telemetry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(nexus_example example.cpp)
target_link_libraries(nexus_example PRIVATE Threads::Threads)
add_executable(nexus_queue_test test.cpp)
target_link_libraries(nexus_queue_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(nexus_queue_test PRIVATE /UNDEBUG)
else()
  target_compile_options(nexus_queue_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME spsc_fifo COMMAND nexus_queue_test)
